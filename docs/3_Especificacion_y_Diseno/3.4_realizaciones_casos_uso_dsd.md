#### DSD - agregarItem(idProducto, cantidad)
[Link al contrato](3.3_contratos_de_operacion.md#contrato-agregaritemidproducto-cantidad)

```plantuml
@startuml
skinparam style strictuml
hide footbox

actor Cajero as C
participant "unOwens:Owens" as Owens
collections "productos" as ColProductos
participant "unProducto:Producto" as Prod
participant "unaComanda:Comanda" as Comanda

C -> Owens : 2. agregarItem(idProducto, cantidad)
note over Owens : retorno=false

loop mientras haya productos y retorno = false
    Owens -> ColProductos : unProducto=obtenerSiguiente()
    Owens -> Prod : retorno = isId(idProducto)
end

opt retorno = true
    create participant "unItemComanda:ItemComanda" as Item
    Owens -->> Item : crear()
    Owens -> Item : agregar(unProducto, cantidad)
    Owens -> Comanda : agregarItem(unItemComanda)
    Owens -> Comanda : subtotal = calcularTotal()
end

Owens -->> C : subtotal
@enduml
```

### 3.4.4. DSD del CU-05 Anular Venta
[Link al DSS](3.2_diagramas_secuencia_sistema.md#325-dss-uc-05-anular-venta)

#### DSD - buscarVenta(nroTicket)
[Link al contrato](3.3_contratos_de_operacion.md#contrato-buscarventanroticket-operación-de-consulta)

```plantuml
@startuml
skinparam style strictuml
hide footbox

actor Cajero as C
participant "unOwens:Owens" as Owens
collections "ventas" as ColVentas
participant "unaVenta:Venta" as Venta

C -> Owens : 1. buscarVenta(nroTicket)
note over Owens : retorno=false

loop mientras haya ventas y retorno = false
    Owens -> ColVentas : unaVenta=obtenerSiguiente()
    Owens -> Venta : retorno = isTicket(nroTicket)
end

alt retorno = true
    Owens -->> C : unaVenta
else retorno = false
    create participant "unaExcepcion:Excepcion" as Ex
    Owens -->> Ex : crear("No se encontró la venta")
end
@enduml
```
